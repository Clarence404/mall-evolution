poll_rps_by_status      | sum by (status) (rate(http_server_requests_seconds_count{application="mall-app",uri="/api/orders/{orderNo}/payment"}[30s]))
payment_rps             | sum by (status) (rate(http_server_requests_seconds_count{application="payment-service",uri="/internal/payments/{orderNo}"}[30s]))
amplification           | sum(rate(http_server_requests_seconds_count{application="payment-service",uri="/internal/payments/{orderNo}"}[30s])) / sum(rate(http_server_requests_seconds_count{application="mall-app",uri="/api/orders/{orderNo}/payment"}[30s]))
retry_failures          | sum by (method, exception) (rate(remote_retryable_failures_total{application="mall-app"}[30s]))
retry_exhausted         | sum by (method) (rate(remote_retryable_exhausted_total{application="mall-app"}[30s]))

fault_inflight          | max by (rule) (demo_fault_inflight{application="payment-service"})
