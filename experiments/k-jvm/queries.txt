# 实验 K 从 Prometheus 导出的曲线：名字 | PromQL（导出步长 5 秒）
app_rps                | sum by (uri, status) (rate(http_server_requests_seconds_count{application="mall-app",uri=~"/api/orders.*"}[30s]))
app_mean               | sum by (uri) (rate(http_server_requests_seconds_sum{application="mall-app",uri=~"/api/orders.*"}[30s])) / sum by (uri) (rate(http_server_requests_seconds_count{application="mall-app",uri=~"/api/orders.*"}[30s]))
app_p99                | histogram_quantile(0.99, sum by (le, uri) (rate(http_server_requests_seconds_bucket{application="mall-app",uri=~"/api/orders.*"}[30s])))
cpu                    | max by (application) (process_cpu_usage{application=~"mall-app|mall-gateway"})
alloc_rate             | sum(rate(jvm_gc_memory_allocated_bytes_total{application="mall-app"}[30s]))
gc_pause_rate          | sum by (gc, action, cause) (rate(jvm_gc_pause_seconds_count{application="mall-app"}[30s]))
gc_pause_time          | sum(rate(jvm_gc_pause_seconds_sum{application="mall-app"}[30s]))
gc_pause_max           | max(jvm_gc_pause_seconds_max{application="mall-app"})
heap_used              | sum(jvm_memory_used_bytes{application="mall-app",area="heap"})
live_data              | max(jvm_gc_live_data_size_bytes{application="mall-app"})
hikari_active          | max(hikaricp_connections_active{application="mall-app"})
tomcat_busy            | max(tomcat_threads_busy_threads{application="mall-app"})
