# 实验 F 从 Prometheus 导出的曲线：名字 | PromQL（导出步长 5 秒）。抓取间隔建议 5 秒，见 experiments/observability/
# 详情接口在 mall-app 的 uri 标签是 /api/products/{skuId}；不存在的 SKU 返回 404，按 status 分开看
detail_rps_by_status   | sum by (status) (rate(http_server_requests_seconds_count{application="mall-app",uri="/api/products/{skuId}"}[30s]))
detail_p99             | histogram_quantile(0.99, sum by (le) (rate(http_server_requests_seconds_bucket{application="mall-app",uri="/api/products/{skuId}"}[30s])))
detail_avg             | sum(rate(http_server_requests_seconds_sum{application="mall-app",uri="/api/products/{skuId}"}[30s])) / sum(rate(http_server_requests_seconds_count{application="mall-app",uri="/api/products/{skuId}"}[30s]))
gateway_status         | sum by (status) (rate(http_server_requests_seconds_count{application="mall-gateway"}[30s]))
route_limited          | sum by (route) (rate(gateway_route_limited_total{application="mall-gateway"}[30s]))
db_load_rate           | sum(rate(product_detail_db_load_total{application="mall-app"}[30s]))
db_load_total          | sum(product_detail_db_load_total{application="mall-app"})
l1_gets_rate           | sum by (result) (rate(cache_gets_total{application="mall-app",cache="product_detail"}[30s]))
l1_hit_ratio           | sum(rate(cache_gets_total{application="mall-app",cache="product_detail",result="hit"}[30s])) / sum(rate(cache_gets_total{application="mall-app",cache="product_detail"}[30s]))
l1_size                | max(cache_size{application="mall-app",cache="product_detail"})
l2_rate                | sum by (result) (rate(product_detail_l2_total{application="mall-app"}[30s]))
hikari_active          | max(hikaricp_connections_active{application="mall-app"})
hikari_pending         | max(hikaricp_connections_pending{application="mall-app"})
hikari_max             | max(hikaricp_connections_max{application="mall-app"})
hikari_acquire_avg     | sum(rate(hikaricp_connections_acquire_seconds_sum{application="mall-app"}[30s])) / sum(rate(hikaricp_connections_acquire_seconds_count{application="mall-app"}[30s]))
hikari_acquire_max     | max(hikaricp_connections_acquire_seconds_max{application="mall-app"})
hikari_timeout         | sum(rate(hikaricp_connections_timeout_total{application="mall-app"}[30s]))
tomcat_busy            | max(tomcat_threads_busy_threads{application="mall-app"})
tomcat_max             | max(tomcat_threads_config_max_threads{application="mall-app"})
cpu                    | max by (application) (process_cpu_usage{application=~"mall-app|mall-gateway"})
