# 实验 I 从 Prometheus 导出的曲线：名字 | PromQL（导出步长 5 秒）
app_rps                | sum by (method, uri, status) (rate(http_server_requests_seconds_count{application="mall-app",uri=~"/api/orders.*"}[30s]))
app_mean               | sum by (method, uri) (rate(http_server_requests_seconds_sum{application="mall-app",uri=~"/api/orders.*"}[30s])) / sum by (method, uri) (rate(http_server_requests_seconds_count{application="mall-app",uri=~"/api/orders.*"}[30s]))
app_p99                | histogram_quantile(0.99, sum by (le, method, uri) (rate(http_server_requests_seconds_bucket{application="mall-app",uri=~"/api/orders.*"}[30s])))
gateway_mean           | sum(rate(http_server_requests_seconds_sum{application="mall-gateway"}[30s])) / sum(rate(http_server_requests_seconds_count{application="mall-gateway"}[30s]))
client_mean            | sum by (application, uri, method) (rate(http_client_requests_seconds_sum{application=~"mall-app|inventory-service"}[30s])) / sum by (application, uri, method) (rate(http_client_requests_seconds_count{application=~"mall-app|inventory-service"}[30s]))
inventory_mean         | sum by (uri) (rate(http_server_requests_seconds_sum{application="inventory-service"}[30s])) / sum by (uri) (rate(http_server_requests_seconds_count{application="inventory-service"}[30s]))
hikari_active          | max by (application) (hikaricp_connections_active{application=~"mall-app|inventory-service"})
hikari_usage_avg       | sum by (application) (rate(hikaricp_connections_usage_seconds_sum[30s])) / sum by (application) (rate(hikaricp_connections_usage_seconds_count[30s]))
hikari_usage_rate      | sum by (application) (rate(hikaricp_connections_usage_seconds_count[30s]))
hikari_acquire_avg     | sum by (application) (rate(hikaricp_connections_acquire_seconds_sum[30s])) / sum by (application) (rate(hikaricp_connections_acquire_seconds_count[30s]))
cpu                    | max by (application) (process_cpu_usage{application=~"mall-app|inventory-service|mall-gateway|payment-service"})
alloc_rate             | sum by (application) (rate(jvm_gc_memory_allocated_bytes_total{application=~"mall-app|inventory-service|mall-gateway"}[30s]))
gc_pause_rate          | sum by (application, gc, action) (rate(jvm_gc_pause_seconds_count{application=~"mall-app|inventory-service|mall-gateway"}[30s]))
gc_pause_time          | sum by (application) (rate(jvm_gc_pause_seconds_sum{application=~"mall-app|inventory-service|mall-gateway"}[30s]))
gc_pause_max           | max by (application) (jvm_gc_pause_seconds_max{application=~"mall-app|inventory-service|mall-gateway"})
tomcat_busy            | max by (application) (tomcat_threads_busy_threads{application=~"mall-app|inventory-service"})
