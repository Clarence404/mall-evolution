# 实验 H 从 Prometheus 导出的曲线：名字 | PromQL（导出步长 5 秒）。H3 有多个 mall-app 实例，抓取目标要包含 18180、18280（见 experiments/observability/）
rps_by_status          | sum by (method, status) (rate(http_server_requests_seconds_count{application="mall-app",uri="/api/orders"}[30s]))
rps_by_instance        | sum by (instance) (rate(http_server_requests_seconds_count{application="mall-app",uri="/api/orders"}[30s]))
p99                    | histogram_quantile(0.99, sum by (le, method) (rate(http_server_requests_seconds_bucket{application="mall-app",uri="/api/orders"}[30s])))
gateway_status         | sum by (status) (rate(http_server_requests_seconds_count{application="mall-gateway"}[30s]))
hikari_active          | max by (instance) (hikaricp_connections_active{application="mall-app"})
hikari_pending         | max by (instance) (hikaricp_connections_pending{application="mall-app"})
hikari_max             | max by (instance) (hikaricp_connections_max{application="mall-app"})
hikari_acquire_avg     | sum by (instance) (rate(hikaricp_connections_acquire_seconds_sum{application="mall-app"}[30s])) / sum by (instance) (rate(hikaricp_connections_acquire_seconds_count{application="mall-app"}[30s]))
hikari_acquire_max     | max by (instance) (hikaricp_connections_acquire_seconds_max{application="mall-app"})
hikari_usage_avg       | sum by (instance) (rate(hikaricp_connections_usage_seconds_sum{application="mall-app"}[30s])) / sum by (instance) (rate(hikaricp_connections_usage_seconds_count{application="mall-app"}[30s]))
hikari_timeout         | sum by (instance) (rate(hikaricp_connections_timeout_total{application="mall-app"}[30s]))
tomcat_busy            | max by (instance) (tomcat_threads_busy_threads{application="mall-app"})
tomcat_max             | max by (instance) (tomcat_threads_config_max_threads{application="mall-app"})
inventory_hikari       | max(hikaricp_connections_active{application="inventory-service"})
inventory_hikari_pending | max(hikaricp_connections_pending{application="inventory-service"})
reserve_server_p99     | histogram_quantile(0.99, sum by (le) (rate(http_server_requests_seconds_bucket{application="inventory-service",uri="/internal/inventory/reserve"}[30s])))
bulkhead_available     | min by (instance, name) (resilience4j_bulkhead_available_concurrent_calls{application="mall-app"})
cb_state               | max by (name, state) (resilience4j_circuitbreaker_state{application="mall-app"})
cpu                    | max by (application, instance) (process_cpu_usage{application=~"mall-app|inventory-service|mall-gateway"})
